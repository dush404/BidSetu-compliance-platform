    49→function CheckIcon({ status }: { status: CheckStatus }) {
    50→  const cls = {
    51→    PASS: "text-ok",
    52→    REVIEW: "text-warn",
    53→    FAIL: "text-bad",
    54→    NA: "text-muted-foreground/50",
    55→  }[status];
    56→  const Icon = {
    57→    PASS: CheckCircle2,
    58→    REVIEW: AlertTriangle,
    59→    FAIL: XCircle,
    60→    NA: CircleMinus,
    61→  }[status];
    62→  return <Icon className={cn("size-4.5 shrink-0", cls)} strokeWidth={status === "NA" ? 1.5 : 2} />;
    63→}
    64→
    65→const badge: Record<CheckStatus, { label: string; cls: string }> = {
    66→  PASS: { label: "Pass", cls: "text-ok bg-ok-soft border-ok/25" },
    67→  REVIEW: { label: "Review", cls: "text-warn bg-warn-soft border-warn/30" },
    68→  FAIL: { label: "Fail", cls: "text-bad bg-bad-soft border-bad/25" },
    69→  NA: { label: "N/A", cls: "text-muted-foreground bg-muted border-border" },
    70→};

   157→            </span>
   158→            <span className="inline-flex items-center gap-1.5 font-mono text-xs">
   159→              GSTIN {bidder.gstin}

   307→                                  "mx-5 mb-4 rounded-lg border p-4 text-[13px] leading-relaxed",
   308→                                  c.status === "FAIL"
   309→                                    ? "border-bad/20 bg-bad-soft/50 text-bad"

   392→            {/* ——— Right rail ——— */}
