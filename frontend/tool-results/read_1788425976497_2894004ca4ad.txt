    21→def load_env_file() -> None:
    22→    """Load environment variables from the local .env file if present."""
    23→    env_file = Path(__file__).resolve().with_name(".env")
    24→    if not env_file.exists():
    25→        return
    26→
    27→    for line in env_file.read_text(encoding="utf-8").splitlines():
    28→        line = line.strip()
    29→        if not line or line.startswith("#") or "=" not in line:
    30→            continue
    31→
    32→        key, value = line.split("=", 1)
    33→        key = key.strip()
    34→        value = value.strip().strip('"').strip("'")
    35→        os.environ.setdefault(key, value)
    36→
    37→
    38→load_env_file()
    39→
    40→STRAPI_URL = os.getenv("STRAPI_URL", "http://localhost:1337/api").rstrip("/")
    41→API_TOKEN = os.getenv("STRAPI_API_TOKEN", os.getenv("STRAPI_API_KEY", ""))
    42→
    43→HEADERS = {
    44→    "Authorization": f"Bearer {API_TOKEN}",
    45→    "Content-Type": "application/json",
    46→}
    47→
    48→
    49→def post(endpoint: str, data: dict[str, Any]) -> dict[str, Any] | None:
    50→    resp = requests.post(f"{STRAPI_URL}/{endpoint}", json={"data": data}, headers=HEADERS, timeout=20)
    51→    if resp.status_code >= 400:
    52→        print(f"ERROR posting to {endpoint}: {resp.status_code} — {resp.text[:200]}")
    53→        return None
    54→    created = resp.json().get("data", {})
    55→    name = (data.get("title") or data.get("gstin") or data.get("udyamNumber") or
    56→            data.get("entityName") or data.get("bidderName") or data.get("establishmentCode") or
    57→            data.get("dpiitNumber") or data.get("nsicNumber") or data.get("esicCode") or
    58→            data.get("panNumber") or "record")
    59→    print(f"  ✓ {endpoint}: id={created.get('id')} — {name}")
    60→    return created
    61→
    62→
    63→def create_all(endpoint: str, records: list[dict]) -> list[int]:
    64→    """Create multiple records and return their IDs."""
    65→    ids = []
    66→    for rec in records:
    67→        created = post(endpoint, rec)
    68→        if created:
    69→            ids.append(created.get("id"))
    70→    return ids
