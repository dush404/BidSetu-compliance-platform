    50→function CheckIcon({ status }: { status: CheckStatus }) {
    51→  const cls = {
    52→    PASS: "text-ok",
    53→    REVIEW: "text-warn",
    54→    FAIL: "text-bad",
    55→    NA: "text-muted-foreground/50",
    56→  }[status];
    57→  const Icon = {
    58→    PASS: CheckCircle2,
    59→    REVIEW: AlertTriangle,
    60→    FAIL: XCircle,
    61→    NA: CircleMinus,
    62→  }[status];
    63→  return <Icon className={cn("size-4.5 shrink-0", cls)} strokeWidth={status === "NA" ? 1.5 : 2} />;
    64→}
    65→
    66→const badge: Record<CheckStatus, { label: string; cls: string }> = {
    67→  PASS: { label: "Pass", cls: "text-ok bg-ok-soft border-ok/25" },

    69→  FAIL: { label: "Fail", cls: "text-bad bg-bad-soft border-bad/25" },
    70→  NA: { label: "N/A", cls: "text-muted-foreground bg-muted border-border" },
    71→};

   157→              <Fingerprint className="size-3.5" /> Contact: {bidder.contactName}
   158→            </span>
   159→            <span className="inline-flex items-center gap-1.5 font-mono text-xs">
   160→              GSTIN {bidder.gstin}
   161→            </span>

   308→                                  "mx-5 mb-4 rounded-lg border p-4 text-[13px] leading-relaxed",
   309→                                  c.status === "FAIL"
   310→                                    ? "border-bad/20 bg-bad-soft/50 text-bad"

   323→                                    </span>
   324→                                  </p>

   327→                                  <span className="text-[10px] font-semibold tracking-[0.14em] text-muted-foreground uppercase">

   332→                                <p className="mt-2.5 flex items-center gap-1.5 text-[11px] text-muted-foreground">
   333→                                  <Database className="size-3" /> {c.source}

   355→                  <Reveal delay={0.05}>

   358→                        <FileDown className="size-4.5" />

   396→            {/* ——— Right rail ——— */}
